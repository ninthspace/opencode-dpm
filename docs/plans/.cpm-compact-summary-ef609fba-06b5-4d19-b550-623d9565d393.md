function asServerFn($) {                    // Gy
  if (isFunction($)) return $;

  if (!isFunction($.server)) return;
  return $.server;
}
function everyExport(mod) {                  // Wy
  const seen = new Set(); const out = [];
  for (const value of Object.values(mod)) {
    if (seen.has(value)) continue;
    seen.add(value);
    const fn = asServerFn(value);
    if (!fn) throw TypeError("Plugin export is not a function");
    out.push(fn);
  }
  return out;
}
async function load(entry, input, sink) {    // Ky
